# What scripts/smolvm/sandbox does around a homelab machine (design D7). The
# hooks run as brian, the pack's user; guest-start.sh (the flight recorder
# and swap) needs root, through brian's passwordless sudo. It runs at every
# start, as the sandbox runs it, and is idempotent.
hooks:
  after_create: |
    set -eu
    install -m 600 /opt/homelab-inputs/vm-age/keys.txt /home/brian/.config/sops/age/keys.txt
    sudo /opt/homelab-sandbox/guest-start.sh 1000 --swap-gib 8
    HOMELAB_CACHE_HOME=/home/brian/.cache VMHOST_CACHE=/home/brian/.cache/homelab-vmhost \
      VMHOST_REGISTRY_CACHE='${STAGE0_HOMELAB_REGISTRY_CACHE}' scripts/vmhost/prepare.sh
  before_run: |
    sudo /opt/homelab-sandbox/guest-start.sh 1000 --swap-gib 8
  timeout_ms: 1800000
